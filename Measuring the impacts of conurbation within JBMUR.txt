// (1) Analysis 1: Calculating NDVI and NDBI
// 1.1 Build Function for Cloud Masking
var landsat_mask = function(image) {

  var cloudShadowBitMask = (1 << 4);  
  var cloudsBitMask = (1 << 3);       
  var qa = image.select('QA_PIXEL');  
  var mask = qa.bitwiseAnd(cloudShadowBitMask).eq(0)  
    .and(qa.bitwiseAnd(cloudsBitMask).eq(0));         
    
    return image.updateMask(mask);
};

// 1.2 Landsat Imagery for each year
// 1.2.1 Landsat5 Imagery for 2000
var landsat5_rgb_2000 = ee.ImageCollection('LANDSAT/LT05/C02/T1_L2')
  .filterDate('2000-01-01', '2001-01-01')
  .map(landsat_mask)
  .median()
  .clip(admin)
  .select(['SR_B3', 'SR_B2', 'SR_B1', 'SR_B4', 'SR_B5', 'SR_B7'])
  .multiply(0.0000275).add(-0.2);

// 1.3.1 Prepare NDVI Calculation for each year

        '1: Water',

        '3: Low vegetation',
        '4: Dense vegetation')
        
var class_lulc = ['Built-up', 'Water', 'Barren land', 'Low vegetation', 'Dense vegetation'];
var palette_lulc = ['FF2400', '0F52BA', 'E5AA70', '7CFC00', '008000'];

// 2.2.2 Training Samples, Random Forest Algorithm, and Calculate LULC area for 2000
var ts_2000 = UR_00
  .merge(WA_00)
  .merge(BL_00)
  .merge(LV_00)
  .merge(DV_00);

var bands_2000 = ['SR_B3', 'SR_B2', 'SR_B1', 'SR_B4', 'SR_B5', 'SR_B7', 'NDVI', 'NDBI'];

var signature_2000 = landsat5_indices_2000.select(bands_2000).sampleRegions({
  collection: ts_2000,
  properties: ['lc'],
  scale: 30
}); 

var randomforest_2000 = ee.Classifier.smileRandomForest(100).train({
  features: signature_2000,
  classProperty: 'lc',
  inputProperties: bands_2000
}); 

var lulc_2000 = landsat5_indices_2000.select(bands_2000).classify(randomforest_2000);

var area_lulc_2000 = ee.Image.pixelArea().divide(1000000)
  .addBands(lulc_2000)
  .reduceRegion({
    reducer: ee.Reducer.sum().group({
      groupField: 1,
      groupName: 'LULC Classification',
    }),
    geometry: admin,
    scale: 30,
    maxPixels: 10e10
  });

print('Area (in km2) for each type of LULC in 2000', area_lulc_2000);

var groupList = ee.List(area_lulc_2000.get('groups'));
var fc = ee.FeatureCollection(groupList.map(function(item) {
  var dict = ee.Dictionary(item);
  var classId = dict.get('LULC Classification');
  var areaVal = dict.get('sum');
  var className = ee.List(class_lulc).get(ee.Number(classId));
  
  return ee.Feature(null, {
    'class_name': className,
    'area': areaVal
  });
}));

var chart_2000 = ui.Chart.feature.byFeature({
  features: fc,
  xProperty: 'class_name',
  yProperties: ['area']
})
.setChartType('PieChart')
.setOptions({
  title: 'JBMUR LULC Area Distribution in 2000',
  colors: palette_lulc,
  sliceVisibilityThreshold: 0
});

print(chart_2000);
